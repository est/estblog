js转换华氏温度到摄氏温度

Posted 2006-12-29 | archive

function f2c(s) {
  var test = /(\d+(\\.\d*)?)F\b/g;    // 初始化模式。
  return(s.replace
    (test,
      function($0,$1,$2) {
        return((($1-32) * 5/9) + "C");
      }
    )
  );
}
document.write(f2c("Water freezes at 32F and boils at 212F."));
