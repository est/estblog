用python写MIDI电子琴

Posted 2007-08-23 | archive

想起了VB时代调用 winmm.dll 制作电子琴，觉得比较好玩，于是用python也写了一个电子琴的原型。有时间在用wx做个GUI吧 :)

#!/usr/bin/env python
#coding:UTF-8

From ctypes import *
import time

hMidiOut=c_long() #系统给MIDI请求分配一个句柄
dMidi=windll.winmm #调用的 winmm.dll
dMidi.midiOutOpen(byref(hMidiOut), -1, 0,0,0) #打开一个MIDI输出设备，句柄为hMidiOut

baseNote = 60 #这个应该是C调
volume = 127 #音量最大
channel = 0 #MIDI的第一个乐器：piano，改成9就是drums

scale = (0, 2, 4, 5, 7, 9, 11, 12, 11, 9, 7, 5, 4, 2, 0)
print scale
for note in scale:
# play a note
midimsg = 0x90 + ((baseNote + note) * 0x100) + (volume * 0x10000) + channel
rc = dMidi.midiOutShortMsg (hMidiOut, midimsg)
time.sleep(0.25)
# turn it off
midimsg = 0x80 + ((baseNote + note) * 0x100) + channel
rc = dMidi.midiOutShortMsg (hMidiOut, midimsg)

dMidi.midiOutClose(hMidiOut)
